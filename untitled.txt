    <script>
        // Variabel pelacak status pembacaan
        let readCount = 0;
        const totalLetters = 3;

        // 1. Fungsi untuk membaca surat
        function toggleLetter(card) {
            const isOpen = card.classList.contains('open');
            
            // Logika: Jika surat ini belum terbuka saat diklik, maka anggap user "membacanya"
            if (!isOpen) {
                // Cek apakah kartu ini sudah pernah dihitung sebelumnya (menggunakan data attribute)
                if (!card.dataset.read) {
                    readCount++;
                    card.dataset.read = "true"; // Tandai bahwa kartu ini sudah dibaca
                    createHeartExplosion(card); // Efek buka
                    checkCompletion(); // Cek apakah semua sudah dibaca
                }
                card.classList.add('open');
            } else {
                card.classList.remove('open');
            }
        }

        // 2. Cek apakah semua surat sudah dibaca
        function checkCompletion() {
            if (readCount === totalLetters) {
                revealProposal();
            }
        }

        // 3. Menampilkan pertanyaan (Unlock Content)
        function revealProposal() {
            // Sembunyikan prompt scroll
            const prompt = document.getElementById('scrollPrompt');
            if(prompt) prompt.classList.add('hidden');

            // Tampilkan section proposal
            const proposal = document.getElementById('proposalSection');
            proposal.classList.add('visible');

            // Scroll otomatis ke bawah agar user melihat kejutannya
            setTimeout(() => {
                proposal.scrollIntoView({ behavior: 'smooth' });
                createFinalConfetti(); // Ledakan hati besar saat muncul
            }, 600);
        }

        // Efek ledakan hati kecil saat membuka surat
        function createHeartExplosion(element) {
            const rect = element.getBoundingClientRect();
            const centerX = rect.left + rect.width / 2;
            const centerY = rect.top;

            for (let i = 0; i < 15; i++) {
                const heart = document.createElement('div');
                heart.classList.add('mini-heart');
                heart.innerText = ['❤️', '💕', '💗', '🥰', '😍'][Math.floor(Math.random() * 5)];
                document.body.appendChild(heart);

                heart.style.left = `${centerX}px`;
                heart.style.top = `${centerY}px`;

                const angle = Math.random() * Math.PI;
                const velocity = 50 + Math.random() * 100;
                const tx = Math.cos(angle) * velocity * (Math.random() > 0.5 ? 1 : -1);
                const ty = Math.sin(angle) * velocity - 50;

                heart.style.setProperty('--tx', `${tx}px`);
                heart.style.setProperty('--ty', `${ty}px`);

                setTimeout(() => heart.remove(), 800);
            }
        }

        // Efek Confetti saat "Wanna be my girlfriend?" muncul
        function createFinalConfetti() {
            const proposalSection = document.getElementById('proposalSection');
            const rect = proposalSection.getBoundingClientRect();
            
            // Buat banyak hati melayang naik dari bawah section ini
            for(let i=0; i<30; i++) {
                setTimeout(() => {
                    const heart = document.createElement('div');
                    heart.classList.add('mini-heart'); // Pakai style yang sama
                    heart.innerText = ['💖', '💗', '🩷', '❤️', '🥰'][Math.floor(Math.random() * 5)];
                    document.body.appendChild(heart);

                    const randomX = rect.left + Math.random() * rect.width;
                    heart.style.left = `${randomX}px`;
                    heart.style.top = `${rect.bottom - 50}px`;

                    const tx = (Math.random() - 0.5) * 100;
                    const ty = -(Math.random() * 300 + 100); // Naik ke atas tinggi

                    heart.style.setProperty('--tx', `${tx}px`);
                    heart.style.setProperty('--ty', `${ty}px`);

                    setTimeout(() => heart.remove(), 800);
                }, i * 100);
            }
        }

        // Efek Hati saat Scroll
        let isScrolling = false;
        window.addEventListener('scroll', () => {
            if (!isScrolling) {
                window.requestAnimationFrame(() => {
                    createScrollHeart();
                    isScrolling = false;
                });
                isScrolling = true;
            }
        });

        function createScrollHeart() {
            // Hanya munculkan hati scroll jika proposal section sudah terlihat
            const proposal = document.getElementById('proposalSection');
            const isVisible = proposal.classList.contains('visible');

            if(isVisible && Math.random() > 0.85) {
                const heart = document.createElement('div');
                heart.classList.add('scroll-heart');
                heart.innerHTML = ['💖', '💗', '🩷', '❤️'][Math.floor(Math.random() * 4)];
                heart.style.left = Math.random() * 100 + 'vw';
                const size = Math.random() * 20 + 10;
                heart.style.fontSize = size + 'px';
                document.body.appendChild(heart);
                setTimeout(() => heart.remove(), 3000);
            }
        }

        // Generate background hati statis saat load
        window.onload = () => {
            const bgContainer = document.getElementById('bg-container');
            for(let i=0; i<15; i++) {
                const heart = document.createElement('div');
                heart.classList.add('bg-heart');
                heart.innerHTML = '❤️';
                heart.style.left = Math.random() * 100 + 'vw';
                heart.style.animationDuration = (Math.random() * 10 + 10) + 's';
                heart.style.animationDelay = (Math.random() * 5) + 's';
                heart.style.fontSize = (Math.random() * 30 + 20) + 'px';
                bgContainer.appendChild(heart);
            }
        };
    </script>
